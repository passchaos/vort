Build named-best benchmark executables without using the calling worktree's
dirty build.zig:

  for side in baseline candidate; do
    if [ "$side" = baseline ]; then ROOT=$BASELINE_ROOT; else ROOT=$CANDIDATE_ROOT; fi
    zig build-exe -O ReleaseFast --dep vort --dep benchmark_provenance \
      -Mroot="$ROOT/tools/zlib_encode_bench.zig" \
      "$EPOCH/record-optimal.o" "$EPOCH/zstd-huffman-x2.o" \
      --dep vail_archive_crypto -Mvort="$ROOT/src/root.zig" \
      -O ReleaseFast -Mbenchmark_provenance="$ROOT/tools/benchmark_provenance.zig" \
      -O ReleaseFast -Mvail_archive_crypto="$VAIL" -lc \
      -femit-bin="$EPOCH/bench-$side"
  done

Build numeric-level executables and the candidate wire probe:

  for side in baseline candidate; do
    if [ "$side" = baseline ]; then ROOT=$BASELINE_ROOT; else ROOT=$CANDIDATE_ROOT; fi
    zig build-exe -O ReleaseFast --dep vort \
      -Mroot="$EVIDENCE/numeric_bench.zig" \
      "$EPOCH/record-optimal.o" "$EPOCH/zstd-huffman-x2.o" \
      --dep vail_archive_crypto -Mvort="$ROOT/src/root.zig" \
      -O ReleaseFast -Mvail_archive_crypto="$VAIL" -lc \
      -femit-bin="$EPOCH/numeric-$side"
  done
  zig build-exe -O ReleaseFast --dep vort \
    -Mroot="$EVIDENCE/wire_probe.zig" \
    "$EPOCH/record-optimal.o" "$EPOCH/zstd-huffman-x2.o" \
    --dep vail_archive_crypto -Mvort="$CANDIDATE_ROOT/src/root.zig" \
    -O ReleaseFast -Mvail_archive_crypto="$VAIL" -lc \
    -femit-bin="$EPOCH/wire-candidate"
